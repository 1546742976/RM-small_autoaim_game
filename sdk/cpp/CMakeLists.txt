cmake_minimum_required(VERSION 3.16)
project(rm_control LANGUAGES CXX)
option(RM_WITH_OPENCV "Build the OpenCV algorithm integration example" ON)
add_library(rm_control src/client.cpp)
target_compile_features(rm_control PUBLIC cxx_std_17)
target_include_directories(rm_control PUBLIC include third_party)
if(WIN32)
  target_link_libraries(rm_control PRIVATE ws2_32)
endif()
add_executable(rm_control_probe examples/probe.cpp)
target_link_libraries(rm_control_probe PRIVATE rm_control)
if(RM_WITH_OPENCV)
  # Some static OpenCV packages export this header-only dependency.
  find_package(Eigen3 QUIET)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgcodecs highgui)
  add_executable(rm_vision_template examples/vision_template.cpp)
  target_include_directories(rm_vision_template PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(rm_vision_template PRIVATE rm_control ${OpenCV_LIBS})
endif()
enable_testing()
add_executable(rm_camera_test tests/camera_test.cpp)
target_link_libraries(rm_camera_test PRIVATE rm_control)
add_test(NAME camera_geometry COMMAND rm_camera_test)
if(MINGW)
  target_link_options(rm_control_probe PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(rm_camera_test PRIVATE -static-libgcc -static-libstdc++)
endif()
